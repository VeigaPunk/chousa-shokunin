+# ---- grep pin (round-35 C1) ------------------------------------------------
+# The matcher shells out to grep(1). Resolving bare `grep` through PATH makes
+# the gate's behaviour a function of the caller's environment: a shim earlier
+# on PATH (this host's interactive shells carry a `pi-uu-grep 0.2.0` uutils
+# grep as a shell builtin/function) answers with the same output while running
+# different code, and a PATH without grep cannot run the gate at all. The gate
+# resolves ONE binary before it evaluates a record, verifies its self-reported
+# provenance (the contract needs GNU grep's -z and -P), and every grep call in
+# this shell runs that absolute path through the wrapper below. An override
+# (RESEARCH_GATE_GREP) must be an absolute path and must pass the same check;
+# a pin that cannot be verified is refused (exit 2, GREP_PROVENANCE) before any
+# record is read, never silently downgraded. CHOUSA_GREP_PIN_ECHO=1 prints
+# `GATE_GREP: path=... source=default|override version="..."` for receipts; it
+# is off by default, so every existing mode keeps its byte-identical output.
+# Provenance is self-reported: a binary that lies about its version is out of
+# this check's reach (round-35 claims.md, C1 limit).
+GREP_BIN=''
+GREP_PIN_SOURCE=''
+GREP_PIN_VERSION=''
+grep() { command "$GREP_BIN" "$@"; }
+
+grep_pin_refuse() { # REASON [CANDIDATE]
+  printf 'research-gate: GREP_PROVENANCE refused: %s (candidate=%s)\n' "$1" "${2:-<none>}" >&2
+  exit 2
+}
+
+grep_pin_verify() { # PATH -> rc0 when the binary reports GNU grep
+  local cand=$1 v
+  [[ -f $cand && -x $cand ]] || return 1
+  v=$("$cand" --version 2>/dev/null | head -n 1 || true)
+  GREP_PIN_VERSION=$v
+  case "$v" in
+    *"GNU grep"*) return 0 ;;
+  esac
+  return 1
+}
+
+grep_pin_init() {
+  local cand='' c=''
+  if [[ -n ${RESEARCH_GATE_GREP:-} ]]; then
+    cand=${RESEARCH_GATE_GREP}
+    GREP_PIN_SOURCE=override
+    case "$cand" in
+      /*) ;;
+      *) grep_pin_refuse "override is not an absolute path" "$cand" ;;
+    esac
+    [[ -f $cand ]] || grep_pin_refuse "override is not a regular file" "$cand"
+    [[ -x $cand ]] || grep_pin_refuse "override is not executable" "$cand"
+    grep_pin_verify "$cand" \
+      || grep_pin_refuse "override does not report GNU grep: ${GREP_PIN_VERSION:-<no version>}" "$cand"
+  else
+    GREP_PIN_SOURCE=default
+    for c in /usr/bin/grep /bin/grep; do
+      if grep_pin_verify "$c"; then cand=$c; break; fi
+    done
+    if [[ -z $cand ]]; then
+      c=$(command -v grep 2>/dev/null || true)
+      case "$c" in
+        /*) if grep_pin_verify "$c"; then cand=$c; fi ;;
+      esac
+    fi
+    [[ -n $cand ]] || grep_pin_refuse "no GNU grep found at /usr/bin/grep, /bin/grep or on PATH" ''
+  fi
+  GREP_BIN=$cand
+  if [[ ${CHOUSA_GREP_PIN_ECHO:-0} == 1 ]]; then
+    printf 'GATE_GREP: path=%s source=%s version="%s"\n' "$GREP_BIN" "$GREP_PIN_SOURCE" "$GREP_PIN_VERSION" >&2
+  fi
+}

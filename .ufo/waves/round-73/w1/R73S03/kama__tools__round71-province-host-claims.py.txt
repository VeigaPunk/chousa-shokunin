156: def parse_markers(text):
157:     """Return (marker paragraphs, {tag: first paragraph date})."""
158:     lines = text.splitlines()
159:     markers = 0
160:     tags = {}
161:     i = 0
162:     while i < len(lines):
163:         match = MARKER_OPEN_RE.match(lines[i])
164:         if not match:
165:             i += 1
166:             continue
167:         markers += 1
168:         date = match.group(1)
169:         block = [lines[i]]
170:         j = i + 1
171:         while j < len(lines) and lines[j].strip() and lines[j][0] in " \t":
172:             block.append(lines[j])
173:             j += 1
174:         for tag in TAG_RE.findall("\n".join(block)):
175:             tags.setdefault(tag, date)
176:         i = j
177:     return markers, tags

180: def marker_paragraphs(text):
181:     """Map each province-host tag to the normalized text of its marker paragraph."""
182:     out = {}
183:     current = None
184:     buf = []

186:     def flush():
187:         if current is None:
188:             return
189:         joined = " ".join(part.strip() for part in buf if part.strip())
190:         for tag in re.findall(r"province-host:([A-Za-z0-9._-]+)", joined):
191:             out.setdefault(tag, joined)

198:             buf.append(line)

353: def covering_correction(claim, corrections):
354:     for correction in corrections:

365: def evaluate(corrections_path, scratch_root):
366:     text = read_text(corrections_path)
367:     markers, marker_tags = parse_markers(text)
368:     corrections = parse_corrections(text)
369:     claims = []
370:     for tag, date in marker_tags.items():
371:         if tag not in SURFACE_MAP:
372:             raise ProbeError(
373:                 f"marker tag province-host:{tag} has no declared probe (add it to SURFACES)")
374:         verdict, command, evidence = probe_surface(tag)
375:         claims.append({"tag": tag, "kind": "marker", "date": date, "claim": "absent",
376:                        "live": verdict, "command": command, "evidence": evidence})
377:     for tag, verdict, prose, probe in EXTRAS:
378:         if probe is None:
379:             live, command, evidence = probe_ntfs_path(scratch_root)
380:         else:
381:             live, command, evidence = probe()
382:         claims.append({"tag": tag, "kind": "extra", "date": MARKER_DATE,

384:                        "live": live, "command": command, "evidence": evidence})

387:         correction = covering_correction(claim, corrections)
388:         claim["covered"] = correction is not None
389:         claim["correction"] = correction

391:     uncovered = sum(1 for claim in claims if claim["contradicted"] and not claim["covered"])
392:     return {"markers": markers, "tags": len(marker_tags), "extra": len(EXTRAS),
393:             "contradicted": contradicted, "uncovered": uncovered,
394:             "claims": claims, "corrections": len(corrections)}

546: def drop_marker_block(text, tag):
547:     lines = text.splitlines(keepends=True)
548:     out = []
549:     i = 0
550:     removed = False
551:     while i < len(lines):
552:         line = lines[i]
553:         match = MARKER_OPEN_RE.match(line.rstrip("\n"))
554:         if match and not removed:
555:             j = i + 1
556:             while j < len(lines) and lines[j].strip() and lines[j][0] in " \t":
557:                 j += 1
558:             block = "".join(lines[i:j])
559:             if tag in block:
560:                 i = j
561:                 removed = True
562:                 continue
563:         out.append(line)
564:         i += 1
565:     if not removed:
566:         raise ProbeError(f"fixture: no marker block names {tag}")

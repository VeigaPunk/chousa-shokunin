def scrub_env(root):
    """Minimal env for fresh legs: PATH from this process, HOME the tree root (no rc/profile reads)."""
    env = {"PATH": os.environ.get("PATH", "/usr/bin:/bin"), "HOME": os.path.abspath(root)}
    return env

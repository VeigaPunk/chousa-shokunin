def main():
    out_lines = []
    killed = 0
    for name, old, new in [(m[0], m[1], m[2]) for m in MUTANTS]:
        n = SRC.count(old)
        if n != 1:
            out_lines.append(f"SETUP {name} applies {n} times (SKIP)")
            continue
        mp = os.path.join(S, f"mut-{name}.py")
        open(mp, "w", encoding="utf-8").write(SRC.replace(old, new))
        p = subprocess.run([sys.executable, "-B", mp, "--selftest"], capture_output=True, text=True)
        out = p.stdout + p.stderr
        failed = "-"
        for ln in out.splitlines():
            if "failed=" in ln or "FAIL" in ln and "cases=" in ln:
                failed = ln
                break
        tail = [ln for ln in out.splitlines() if "SELFTEST" in ln]
        fails = [ln for ln in tail if "FAIL" in ln and "SEARCH_RERUN_SELFTEST" not in ln]
        dead = p.returncode == 1 and "SEARCH_RERUN_SELFTEST: FAIL" in out
        killed += 1 if dead else 0
        out_lines.append(f"MUTANT {name} rc={p.returncode} {'KILLED' if dead else 'SURVIVED'} failed_cases={len(fails)} {failed}")
        if fails:
            out_lines.append("  failing: " + " | ".join(fails))
        else:
            out_lines.append("  failing: (none) lastlines=" + " // ".join(tail[-2:]))
    out_lines.append(f"KILLED {killed}/{len([m for m in MUTANTS])} total-probed={len(MUTANTS)}")
    text = "\n".join(out_lines) + "\n"
    sys.stdout.write(text)
    open(os.path.join(S, "mutprobe-out.txt"), "w").write(text)

if __name__ == "__main__":
    main()

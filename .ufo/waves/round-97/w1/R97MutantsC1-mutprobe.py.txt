def run_selftest(path):
    p = subprocess.run([sys.executable, "-B", path, "--selftest"],
                       capture_output=True, text=True)
    dead = p.returncode == 1 and "SEARCH_RERUN_SELFTEST: FAIL" in p.stdout \
        and "Traceback" not in p.stderr
    failed = next((ln.split("failed=")[1] for ln in p.stdout.splitlines()
                   if "failed=" in ln), "-")
    return dead, p.returncode, failed


def run_r95(mutant_path, tag):
    work = os.path.join(S, "w", tag)
    cmd = [sys.executable, "-B",
           os.path.join(S, "repo", "corpus", "evidence", "round95",
                        "rerun-controls-r95.py"),
           "--root", os.path.join(S, "repo"), "--work", work,
           "--tool", mutant_path]
    p = subprocess.run(cmd, capture_output=True, text=True, timeout=1200)
    last = [ln for ln in p.stdout.splitlines() if "CONTROLS" in ln]
    tally = [ln for ln in p.stdout.splitlines() if ln.startswith("CASE")]
    bad = [ln for ln in tally if "FAIL" in ln]
    return p.returncode, last[0] if last else "(no verdict line)", bad


def main():
    out = []
    killed, survived = 0, []
    for name, old, new in MUTANTS:
        n = SRC.count(old)
        if n != 1:
            out.append(f"SETUP {name} applies {n} times (SKIP)")
            continue
        mp = os.path.join(S, f"mut-{name}.py")
        open(mp, "w", encoding="utf-8").write(SRC.replace(old, new))
        dead, rc, failed = run_selftest(mp)
        verdict = "KILLED" if dead else "SURVIVED"
        out.append(f"MUTANT {name} rc={rc} {verdict} failed={failed}")
        if dead:
            killed += 1
        else:
            survived.append((name, mp))
    out.append(f"SELFTEST killed={killed} survived={len(survived)} "
               f"probed={len(MUTANTS)}")
    for name, mp in survived:
        rc, verdict, bad = run_r95(mp, name)
        cover = "covered-by-r95" if rc != 0 else "UNCOVERED-by-r95"
        out.append(f"R95 {name} rc={rc} {cover} verdict=[{verdict}]")
        for ln in bad[:4]:
            out.append(f"  FAILING: {ln}")
    text = "\n".join(out) + "\n"
    sys.stdout.write(text)
    open(os.path.join(S, "probe-out.txt"), "w").write(text)


if __name__ == "__main__":
    main()

def selftest(path):
    p = subprocess.run([sys.executable, "-B", path, "--selftest"], capture_output=True, text=True, cwd=OUT)
    dead = p.returncode == 1 and "SEARCH_RERUN_SELFTEST: FAIL" in p.stdout and "Traceback" not in p.stderr
    failed = next((ln.split("failed=")[1] for ln in p.stdout.splitlines() if "failed=" in ln), "-")
    last = p.stdout.splitlines()[-1] if p.stdout.splitlines() else ""
    return dead, p.returncode, failed, last, p.stderr.count("Traceback")


def main():
    out, killed, survived = [], 0, []
    for name, old, new in MUTANTS:
        n = SRC.count(old)
        if n != 1:
            out.append(f"SETUP {name} applies {n} times SKIP")
            continue
        mp = os.path.join(OUT, f"mut-{name}.py")
        with open(mp, "w", encoding="utf-8") as fh:
            fh.write(SRC.replace(old, new))
        dead, rc, failed, last, tb = selftest(mp)
        out.append(f"MUTANT {name} rc={rc} {'KILLED' if dead else 'SURVIVED'} failed={failed} "
                   f"tracebacks={tb} last={last!r}")
        if dead:
            killed += 1
        else:
            survived.append(name)
    out.append(f"PROBE killed={killed} survived={len(survived)} applied={sum(1 for l in out if l.startswith('MUTANT'))} "
               f"listed={len(MUTANTS)}")
    out.append("SURVIVORS " + (",".join(survived) or "-"))
    text = "\n".join(out) + "\n"
    sys.stdout.write(text)
    with open(os.path.join(S, "probe98-out.txt"), "w", encoding="utf-8") as fh:
        fh.write(text)


if __name__ == "__main__":
    main()

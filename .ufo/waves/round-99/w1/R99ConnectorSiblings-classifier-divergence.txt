# round 99: interstitial challenge pages (the capture is the gate, not the document). Markers are matched against the
# normalized, lower-cased capture text; each is from an observed capture or a widely served interstitial:
# S229/S021/S022/S032 "cookies must be enabled", S230 "checking your browser" and "recaptcha", S234 "client
# challenge" and "a required part of this site couldn't load"; Cloudflare "just a moment", "verify you are human",
# "enable javascript and cookies to continue", "attention required", "ddos protection"; generic "are you a robot".
CHALLENGE_MAX_WORDS = 200
CHALLENGE_MARKERS = (
    "cookies must be enabled",
    "checking your browser",
    "recaptcha",
    "client challenge",
    "a required part of this site couldn't load",
    "just a moment",
    "verify you are human",
    "enable javascript and cookies to continue",
    "attention required",
    "ddos protection",
    "are you a robot",
)


def is_challenge(text):
    """True when normalized TEXT is a short page carrying a challenge marker (CHALLENGE, round 99)."""
    t = norm(text).lower()
    return len(t.split()) <= CHALLENGE_MAX_WORDS and any(m in t for m in CHALLENGE_MARKERS)

def capture_kind(text: str) -> str:

    low = text[:4000].lower()
    for marker in INTERSTITIAL_MARKERS:

#!/usr/bin/env python3
import sys, types, importlib.util
_spec = importlib.util.spec_from_file_location('_gate', '/fast/scratch/hyaku/chousa-r99/R99CriticClaims/repo/tools/gate-records.py', loader=None)
import importlib.machinery
_loader = importlib.machinery.SourceFileLoader('_gate', '/fast/scratch/hyaku/chousa-r99/R99CriticClaims/repo/tools/gate-records.py')
_spec.loader = _loader
_mod = importlib.util.module_from_spec(_spec)
_loader.exec_module(_mod)

PROD_PATHS = {
    'corpus/captures/b5b8d050eea0c638e95ba674.txt',
    'corpus/captures/b90988847e704a08aabc8321.txt',
    'corpus/captures/0954149536aaad411000e1fb.txt',
}
_orig_check = _mod.check

def check(rec, root, repo, reg, refetch):
    sid, kv, quote, form = rec
    if form == 'admit' and 'fixture.invalid' in kv.get('url', ''):
        if '/n' in kv['url']:
            return ['CHALLENGE']
        return _orig_check(rec, root, repo, reg, refetch)
    if form == 'admit' and kv.get('capture') in PROD_PATHS:
        return ['CHALLENGE']
    if form == 'rev' and kv.get('path') in PROD_PATHS:
        return ['CHALLENGE']
    return _orig_check(rec, root, repo, reg, refetch)

_mod.check = check
sys.exit(_mod.main())

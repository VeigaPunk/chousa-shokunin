#!/usr/bin/env python3
import os, sys
S="/fast/scratch/hyaku/chousa-r99/R99MutantsC1"
src=open(f"{S}/repo/tools/gate-records.py",encoding="utf-8").read()
os.makedirs(f"{S}/mutants",exist_ok=True)

def one(id_, old, new, note):
    n = src.count(old) if old else -1
    if n != 1:
        print(f"MUTGEN {id_} OLD-COUNT={n} (want 1) {'NOTE:'+note if note else ''}")
        return False
    out = src.replace(old,new,1)
    open(f"{S}/mutants/{id_}.py","w",encoding="utf-8").write(out)
    print(f"MUTGEN {id_} ok")
    return True

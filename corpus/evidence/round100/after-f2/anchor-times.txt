# script:
import re,sys,subprocess
rev=sys.argv[1]
src=subprocess.run(["git","show",f"{rev}:corpus/SOURCES.md"],capture_output=True,text=True).stdout if rev!="WORKTREE" else open("corpus/SOURCES.md").read()
ref=open("corpus/evidence/round57/refetch.tsv").read().split("\n")
bad=tot=0
for sid in ("S237","S239"):
    row=next(l for l in src.split("\n") if l.startswith(f"| {sid} |"))
    anchor=row.split(" | ")[5]
    times=re.findall(r"(?:2026-11-09T)?([0-9]{2}:[0-9]{2}:[0-9]{2})Z",anchor)
    logged={l.split("\t")[4][11:19] for l in ref if l.startswith(sid+"\t")}
    for t in times:
        tot+=1; ok=t in logged; bad+=not ok
        print(f"ANCHOR {sid} {t} {'LOGGED' if ok else 'NOT-IN-REFETCH'}")
print(f"ANCHOR_TIMES rev={rev[:8]} logged={tot-bad}/{tot}"); sys.exit(1 if bad else 0)
